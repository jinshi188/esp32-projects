## 第 1 步：板载 LED 呼吸（PWM）

```cpp
#include <Arduino.h>

#define LED_PIN 2

void setup() {
  Serial.begin(115200);
  ledcSetup(0, 5000, 8);        // 通道 0，5kHz，8 位 -> duty 0~255
  ledcAttachPin(LED_PIN, 0);    // 把 GPIO2 绑到通道 0
}

void loop() {
  for (int d = 0; d <= 255; d++) { ledcWrite(0, d); delay(8); }
  for (int d = 255; d >= 0; d--) { ledcWrite(0, d); delay(8); }
}
```

## 第 2 步：改成 millis 非阻塞 + 按键

```cpp
#include <Arduino.h>

#define LED_PIN 2
#define KEY_PIN 4
#define TICK_MS 8
#define DEBOUNCE_MS 25

int      duty = 0;
int      dir  = +1;
uint32_t tTick = 0, tChange = 0;
int      keyRaw = HIGH, keyStable = HIGH;

void setup() {
  Serial.begin(115200);
  ledcSetup(0, 5000, 8);
  ledcAttachPin(LED_PIN, 0);
  pinMode(KEY_PIN, INPUT_PULLUP);
}

void loop() {
  uint32_t now = millis();

  int raw = digitalRead(KEY_PIN);
  if (raw != keyRaw) { keyRaw = raw; tChange = now; }
  if ((now - tChange) >= DEBOUNCE_MS && keyStable != keyRaw) {
    keyStable = keyRaw;
    if (keyStable == LOW) Serial.printf("[%lu ms] 按下了\n", millis());
  }

  if ((now - tTick) >= TICK_MS) {
    tTick = now;
    duty += dir;
    if (duty >= 255)      { duty = 255; dir = -1; }
    else if (duty <= 0)   { duty = 0;   dir = +1; }
    ledcWrite(0, duty);
  }
}
```

## 第 3 步：加外接 LED + 模式切换（成品）

```cpp
#include <Arduino.h>

/* 引脚 */
#define PIN_LED_BOARD 2
#define PIN_LED_EXT   16
#define PIN_KEY       4

/* PWM：ESP32 有 16 个独立 LEDC 通道，每个 LED 各占一个 */
#define CH_BOARD 0
#define CH_EXT   1
#define PWM_FREQ 5000
#define PWM_BITS 8

/* 呼吸节奏 */
#define TICK_MS 8
#define STEP    1

/* 消抖 */
#define DEBOUNCE_MS 25

enum Mode { MODE_BREATH, MODE_ON, MODE_OFF };
const char* MODE_NAME[] = { "呼吸", "长亮", "熄灭" };

Mode mode = MODE_BREATH;
int  duty = 0;
int  dir  = +1;
uint32_t tTick = 0, tChange = 0;
int  keyRaw = HIGH, keyStable = HIGH;

void outputDuty() {
  ledcWrite(CH_BOARD, duty);
  ledcWrite(CH_EXT,   duty);
}

void switchMode() {
  mode = (Mode)((mode + 1) % 3);
  if (mode == MODE_ON)     duty = 255;
  if (mode == MODE_OFF)    duty = 0;
  if (mode == MODE_BREATH) dir  = +1;
  outputDuty();
  Serial.printf("[%8lu ms] 按键 -> 模式：%s（duty=%d）\n",
                (unsigned long)millis(), MODE_NAME[mode], duty);
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println("\n=== 项目一：呼吸灯 + 按键交互小灯 ===");
  Serial.println("按一下切换：呼吸 -> 长亮 -> 熄灭");
  Serial.flush();

  ledcSetup(CH_BOARD, PWM_FREQ, PWM_BITS);
  ledcAttachPin(PIN_LED_BOARD, CH_BOARD);
  ledcSetup(CH_EXT, PWM_FREQ, PWM_BITS);
  ledcAttachPin(PIN_LED_EXT, CH_EXT);

  pinMode(PIN_KEY, INPUT_PULLUP);   // 内部上拉：平时 HIGH，按下 LOW

  outputDuty();
}

void loop() {
  uint32_t now = millis();

  /* 按键消抖 */
  int raw = digitalRead(PIN_KEY);
  if (raw != keyRaw) { keyRaw = raw; tChange = now; }
  if ((now - tChange) >= DEBOUNCE_MS && keyStable != keyRaw) {
    keyStable = keyRaw;
    if (keyStable == LOW) switchMode();
  }

  /* 呼吸渐变（只在呼吸模式跑） */
  if (mode == MODE_BREATH && (now - tTick) >= TICK_MS) {
    tTick = now;
    duty += dir;
    if (duty >= 255)      { duty = 255; dir = -1; }
    else if (duty <= 0)   { duty = 0;   dir = +1; }
    outputDuty();
  }
}
```

## 第 4 步（可选）：让呼吸更像"呼吸"

线性渐变的呼吸看着有点假，真正的呼吸是快的缓入缓出。用正弦：

```cpp
#include <math.h>

// 放在 loop 的呼吸分支里，替换 duty += dir
float phase = (float)(now % 4000) / 4000.0f;          // 0~1，4 秒一个周期
duty = (int)(127.5f * (1.0f - cosf(2.0f * PI * phase)));  // 0~255 正弦起伏
```

> 顺带一提：`sin` 正是你后面端侧 AI 路线里 TFLite Micro 的第一个练手模型（用神经网络拟合 sin 函数）。
> 今天先在 MCU 上看见它，寒假再用模型去"学"它。
